سُمیریہ: تہذیب تہٕ کنٹرول نِظامَن ہُنٛد زینہٕ جاے

عظیم دریافت

تَقریٖبَن 3500 قَبٕل مسیٖح، دٔریاوَن ٹِگرِس تہٕ یوٗفرَٹِس دَرمیان زَرخیٖز زٔمیٖنَس منٛز، کٔر سُمیرینَن اَکھ غٲر معمۄلی چیٖز دٔریافَت۔ تِمو کوٚر مَحسوٗس زِ اِنسانُک دِماغ چھُ حَقیٖقَتُک اَکھ نموٗنہٕ بَناوان، تہٕ اَتھ نموٗنَس قابوٗ کٔرِتھ ہیٚکَن لوٗکھ قابوٗ کَرنہٕ یِتھ۔

اَمہِ دریافتہِ کٔر اِنسٲنی تٲریٖخ دٲیمی طور تبدیل۔ سُمیریو کٔر نہٕ صِرِف تہذیبٕچ گۄڈٕنِچ کَدَم تُل — تِمو کٔر قٲبُوٗک گۄڈٕنِچ ساینس قٲیِم۔ دٔریاو ٹِگرِس تہٕ یوٗفرٹیزس درمیان یہِ زٔمیٖن بَنِیہِ تِمن نِظامَن ہٕنٛز زینہٕ جاے یِمو اِنسٲنی تٲریٖخٕکۍ 5,000 ؤری مُشخص کٔرۍ۔

شہرِ اُر — گۄڈٕنیُک کنٹرول سینٹر

قدیم سُمیریہَس منٛز بنیوو 'عُر' شَہَر اِنسٲنی تٲریخَس منٛز گۄڈٕنِکِس کنٹرول نِظامُک مرکز۔ تَتہِ کٔر پادٕرِ طبقَن پنٛنِس خُفیہ عِلمَس اِستعمال کٔرِتھ لوٗکَن کنٹرول کَرنٕچ طٔریٖقہٕ تیار۔ تِم ٲسۍ نہٕ صِرِف مذہبی رَہنُما — تِم ٲسۍ تٲریٖخ دان، ساینس دان، تہٕ بینک مینیجر۔

زِگورَت — علم، طاقت تہٕ دولتُک مرکز

اُرُک زِگورات اوس نہٕ صِرِف اَکھ مَندر — یہِ اوس عِلم، طاقَت تہٕ دَولتُک مُقام۔ یِمَن عمارتَن مَنٛز ٲسۍ پادٕرِ عٲلِم پنُن خُفیہ عِلم مَحفوٗظ تھاوان، بَڈراوان تہٕ نَسلَن مَنٛز مُنتقِل کَران۔

زِگورَت اوس اَکھ اہرام وُجُود۔ اَمیُک پٮ۪ٹھِم حِصہٕ اوس وَننہٕ یِوان ییٚتہِ دیوتا روزان ٲس۔ مگر اَصٕلَس منٛز اوس یہِ پادٕرِ ہُنٛد خُفیہ مرکز۔ تَتہِ پٮ۪ٹھٕ ٲسۍ تِم پوٗرٕ معاشرَس کنٹرول کَران۔

کنٹرولٕکۍ ژور ستوٗن

سُمیری پادٕرِیو کٔر ژورن اہم شعبن منٛز قٲبلہِ ذِکِر مہارت حٲصِل:

1. فلکیات

گرہن، موسم، تہٕ فلکیاتی چکرن ہنٛز پیشین گوئی کران۔ تٔمۍ وُچھ آسمان نزدیٖکی سۭتۍ تہٕ کٔرٕن آسمٲنی واقعاتن ہنٛد صحیح حساب۔ شمسی یا قمری گرہن کٔرٕن تٔمۍ برٛونٛہہ زانان، تٔمۍ کٔرٕن تِم خدایی نشانی ہٕنٛدۍ پٲٹھۍ پیش تاکہ لوٗکھ کھوژن — ییٚمہِ سۭتۍ تٔمۍ سٕنٛدِ اِختیار مضبوط گٔیہِ۔

2. رِیاضی

پیمائش تہٕ حساب خٲطرٕ جدید نِظام بناوُن۔ سُمیریو کٔر سیکساجیسمل (بیس-60) نمبر نِظام تیار — یُس اَز تہِ وقت (60 سیکنڈ، 60 منٹ) تہٕ دائرٕ (360 ڈگری) ماپانہٕ خٲطرٕ اِستعمال چھُ گژھان۔ اَتھ ریاضیٲتی عِلمَن دِژ تِمن زٔمیٖن ماپانٕچ، ٹیکس جمع کرنٕچ، تہٕ پیچیدٕ ڈھانچہٕ بناونٕچ طاقت۔

3. لیٚکھُن (کیونیفارم)

علم بچاونہٕ خٲطرٕ کیونیفارم رسم الخط تیار کرُن۔ کیونیفارم اوس دُنیاہُک گۄڈنیُک تحریری نِظام۔ مٲژِ ہٕنٛدٮ۪ن تختن سۭتۍ ٲسۍ سُمیری قانوٗن، تجٲرتی معاہدٕ، سائنسی مُشاہِدٕ تہٕ مذہبی متن لیٚکھان۔ تاہم، یہِ تحریری علم اوس زیٛادٕ تر پادٕرۍ طبقس تام مَحدوٗد — عام لوٗکھ ہیٚکَن نہٕ پرنہٕ۔ اَتھ سۭتۍ آیہِ علمٕچ نابرابری مَضبوٗت گژھِتھ۔

4. اِقتِصادِیَت

قرض، سود، تہٕ وسٲئل تقسیم کرُن۔ سمیری مندر ٲسۍ نہٕ صِرِف عبادت گاہہٕ — تِم ٲسۍ دُنیاہٕکۍ گۄڈٕنِکۍ بینک۔ کٲشتر ٲسۍ مندرٕ پٮ۪ٹھٕ بیٲنۍ، جانور، یا رۄپہٕ قرضہٕ نِوان تہٕ فصل کٹنہٕ پتہٕ دِوان سود۔ اَتھ اِقتِصٲدی نِظامہٕ سۭتۍ، پادٕرِ ٲسۍ لوٗکن ہٕنٛدِ زِنٛدگی پٮ۪ٹھ پوٗرٕ اِختیار حٲصِل کَران۔

مندر — ٹرینیٹی کنورجنس

یہِ سٲری عِلم اوس تِہنٛدٮ۪ن منٛدرن منٛز مَحفوٗظ تھاونہٕ آمُت۔ منٛدر اوس نہٕ صِرِف اَکھ مَذہبی مَرکز — یہِ اوس عِلم، دَولت تہٕ طاقتُک مِلَن۔ یہِ اِختیار زِ کٲنٛسہِ کُنہِ عِلم مِلہِ تہٕ کٲنٛسہِ کُنہِ کٲنٛسہِ عِلم، یہِ اوس پادٕرٮ۪ن ہٕنٛدِس اِختیارس منٛز۔

توسیع — علمُک سفر

ییٚلہِ سُمیری تہذیٖب ختم گٔیہِ، یہِ کنٹرول علم آو نہٕ تباہ کرنہٕ — یہِ آو بیٚیِس تہذیٖبَس منٛز منتقل کرنہٕ۔ یہِ گٔیہِ مِصر، وادیِ سِندھ، یونان تہٕ رومَس منٛز۔ پرٛٮ۪تھ تہذیٖبَن کٔر یہِ کنٹرول ماڈل پنٛنِس ثقافتس مُطٲبِق ڈھالُن — مگر بُنیٲدی اُصوٗل روٗدُس یِتھُے: علم کینٛہَن اتھَن منٛز جمع کَرُن تہٕ لوٗکَن ہٕنٛدِس اِحساسَس پٮ۪ٹھ کنٹرول تھاوُن۔

سُمیریو دٔسۍ بناونہٕ آمٕتۍ کنٹرول نِظامو دِژ اِنسانَن ہٕنٛدِس معاشرس ساسہٕ بٔدۍ ؤریَن تام شکل۔ اَزٕکۍ جدیٖد اِدارٕ — بینک، یوٗنیورسٹی، میڈیا تنظیٖم — چھِ اَتھ قدیٖم نموٗنَس ہٕنٛدۍ ارتقٲیی شکلہٕ۔

یہِ چھُ "دُنیاہ کٮ۪ن کنٹرولرن ہُنٛد تٲریٖخ" سیریزُک دوٚیِم مضمون۔ بیٚیِس مضمونَس منٛز پرٛیٖو مصرٕک کنٹرول سسٹم، رازدارانہٕ سکوٗلَن، تہٕ اخیناطن سٕنٛدِس اِنقلابَس مُتعلِق۔
