مِصری نِظام: مذہبی کنٹرول خٲطرٕ بہترین نموٗنہٕ — تہٕ اخیناطن سُنٛد اِنقلاب۔

کمالی کنٹرول ماڈل

سُمیریو ہٕنٛدِ کنٹرول نِظامٕچ بُنیاد تھاونہٕ پتہٕ، یہِ عِلم گوٚو نیل دٔریاو کِس بٔٹھِس تام۔ مگر قدیٖم مِصرَن کٔر یہِ ماڈل پوٗرٕ پٲٹھۍ اَکھ نوٚو سطحَس تام۔ ییٚلہِ زَن سُمیریو عِلمہٕ سۭتۍ کنٹرول ہیٚچھناوان ٲسۍ، مِصریو کٔر مذہب تہٕ الٰہی یٔقیٖن سۭتۍ کنٹرول مَکمل۔ یہِ تہذیٖب یۄس نیل دٔریاوَس أندِ پٔکۍ بَنِیہِ، بنییہِ اِنسانَن ہٕنٛدِس تٲریخَس مَنٛز سارِوٕے کھۄتہٕ زیٛادٕ دیرپا کنٹرول نِظام — 3,000 ؤری کھۄتہٕ زیٛادٕ۔

دۄن سطحن ہُنٛد ڈھانٛچہٕ

مصرِیو بنوو اکھ واریاہ اثر دار دۄن سطحن ہٗند سمٲجی ڈھانچہٕ ۔

گۄڈٕنِچ درجہٕ — عام لوٗکھ:کٲریٖن، کاریگر تہٕ سپاہ۔ تِم ٲسۍ یٔقیٖن تھاوان زِ فرعون اوس اکھ حقیقی دیوتا۔ ییٚلہِ پرٛٮ۪تھ ؤرِیہِ دٔریاو نیلَس سٔہلاب یِوان اوس تہٕ زٔمیٖن زرخیز بناوان اوس، تِم ٲسۍ یٔقیٖن تھاوان زِ یہِ اوس فرعون سٕنٛدِ خدٲیی طاقتہٕ کِنۍ گژھان۔

دوٚیم درجہٕ — پادٕرۍ طبقہٕ:تِمَن اوس اصلی عِلم حٲصِل۔ تِمَن کٔرِ دٔریایہِ نیلہِ ہُنٛد سٔہلاب تہٕ زَرٲعی موسم قابوٗ۔ اگرچہ فرعون باہَرٕ پٲٹھۍ اوس اَکھ دیوتاہ باسان، مگر اَصٕل اِختیار اوس اَتھ پادٕرۍ طبقس نِش۔

فاروہ — اکھ کٹھ پتلی

اَتھ نِظامَس منٛز ٲس نہٕ فرعون اکھ عام حُکمران۔ فرعون اوس پانہٕ خۄدایہٕ — زِنٛدٕ خۄدایہٕ پٲٹھۍ، مُردٕ خۄدایہٕ پٲٹھۍ۔ لوٗکھ ٲسۍ تٔمِس زِنٛدٕ تہٕ مُردٕ پُزاہ کَران۔ تِم ٲسۍ وَنان زِ پوٗرٕ کائنات، زٔمیٖن تہٕ دُنیا ٲس فرعونَس أندِ پٔکۍ گَردِش کَران۔

مگر اَکھ دھوکہٕ اوس۔ فرعون اوس نہٕ اَصٕل حُکمران۔ فرعون اوس صِرِف اَکھ نُمٲئشی سَر۔ تٔمۍ کٔر نہٕ کُنہِ تہِ کٲم۔ سۄ ٲس صِرِف تَتھ بؠہِتھ۔ تٔمۍ کٔر نہٕ نِظامَس پٮ۪ٹھ قوبوٗ۔ واریاہ لوٗکھ چھِ سَمَجان زِ "فرعون چھُ پرٛٮ۪تھ کَتھہِ ہُنٛد نِگرٲنی کَران" مگر یہِ چھُ نہٕ اَصٕل۔ فرعون اوس صِرِف اَکھ کٹھ پُتلی۔ سۄ ٲس کیٚنٛہہ لوٗکن ہٕنٛدِ دٔسۍ قوبوٗ کَرنہٕ یِوان — پادٕرۍ قوبوٗ کَرن وٲلۍ۔ یِم پادٕرۍ قوبوٗ کَرن وٲلۍ ٲسۍ اَصٕلَس منٛز مِصرُک نِظام چلاوان۔

پرٛٮ۪تھ ؤرِیہِ چھُ دٔریاو نیلَس سٔہلاب یِوان۔ اَتھ سٔہلابَس سۭتۍ آبپاشی چھِ نہٕ قُدرتی عملہٕ بلکہٕ فرعون سٕنٛز تخلیقہٕ پٲٹھۍ پیش یِوان کرنہٕ۔ کینہہ وِزِ اوس نہٕ دٔریاو نیلَس سٔہلاب یِوان۔ تیٚلہِ ٲسۍ تِم لوٗکَن وَنان "تۄہہِ چھِو کُچھ غلط کوٚرمُت، اَتھ کِنۍ چھُ فرعون ناراض" تہٕ تِم ٲسۍ یہِ دٔلیٖل دِوان۔

مِسٹری سکوٗل

مصرٕکِس کنٹرول سسٹمُک مرکز اوس مِسٹری سکوٗل۔ یِم ٲسۍ نہٕ صِرِف تٲلیٖمی اِدارٕ — یِم ٲسۍ کنٹرول علمٕکۍ قلعہٕ۔ یِمَن خُفیہ سکوٗلن منٛز ٲسۍ پادٕرۍ نسلن پٮ۪ٹھٕ جمع گژھن وول علم مَحفوٗظ تھاوان۔

اَتھ عِلمَس منٛز کیٛاہ اوس؟ آسمان، ریاضی، فلکیات، قۄدرتی قٲنون، نیل چکر، تعمیٖرٲتی ساینس، جیومیٹری، مُقدس جیومیٹری، تناسب — یہِ سٲری۔ یہِ عِلم اوس نہٕ عام لوٗکَن خٲطرٕ دٔستیاب۔ یہِ اوس صِرِف پادٕرٮ۪ن نِش۔

یِمَن رازدارانہٕ سکوٗلَن مَنٛز اوس پادٕرِ اَتھ عِلمَس قدم بہ قدم، لۄکُٹ لۄکُٹ کٔرِتھ دِنہٕ یِوان۔ پرٛٮ۪تھ مرحلس اوس اَکھ خاص شروعاتی عمل۔ یہِ اوس اَکھ واریاہ پیچیدٕ نِظام۔ کانٛہہ تہِ ہیٚکہِ نہٕ اَکہِ لٹہِ سٲری کَتھ ہیٚچھِتھ۔ قدم بہ قدم، سطح بہ سطح، پرٛٮ۪تھ مرحلس پٮ۪ٹھ اوس اَکھ راز ظٲہِر گژھان۔

نیل سائیکلُک علم

تِم ہیٚکَن صٔحیٖح پٲٹھۍ حِساب کٔرِتھ زِ نیٖلَس کٮ۪س وِزِ ییٚہِ سٔہلاب تہٕ کٮ۪س وِزِ ییٚہِ کم۔ اَمہِ سۭتۍ آو تِمن زٔمیٖنی پٲداوارَس پٮ۪ٹھ پوٗرٕ قوبوٗ۔ عام کِسانَس خٲطرٕ اوس نیٖل اَکھ راز — پادٕرِس خٲطرٕ اوس یہِ اَکھ اوزار۔

تعٲمیرٲتی ساینس تہٕ مقدس جیومیٹری

پِرا مِڈ ٲسۍ نہٕ صِرِف مقبرٕ — تِم ٲسۍ اِختیارٕچ یادگار۔ اَکھ پِرا مِڈ بناونَس خٲطرٕ ٲسۍ ساسہٕ بٔدۍ کٲم کَرن وٲلۍ، دَہٲییَن ہُنٛد وقت تہٕ واریاہ وَسیٖلہٕ درکار۔ یِمَن بٔڑَن ڈھانچَن دِیُت لوٗکَن اَکھ پٲغام: "فاروہ سُنٛد طاقت چھُ اَتھ حَدَس تام زِ سُہ ہیٚکہِ آسمانَس چھوٗن وٲلۍ یادگار بناوِتھ۔" مُقدس جیومیٹری تہٕ تناسبَن سۭتۍ — یُس عِلم صِرِف پادٕرِین ہٮ۪وَن چھُ زانان — آیہِ پِرا مِڈ صٔحیٖح پیمٲیِشَن سۭتۍ بناونہٕ۔

طِبی تہٕ فلکیات

پادٕرِیو کٔر انسٲنی جسمس، بیمارن تہٕ علاجس متعلق وسیع علم حٲصِل۔ شمسی گرہن تہٕ تارکن ہٕنٛز حرکت یِتھۍ آسمٲنی واقعاتن ہٕنٛز پیشین گوئی کٔرِتھ کٔر تِمو لوٗکن یقین دِلاونہٕ زِ تِہنٛدِس دیوتاہن سۭتۍ چھُ اکھ خاص تعلق۔

یہِ کیٛازِ روُد یوٗت واریٛاہ کالس؟

مصرُک نِظام روٗد 3,000 ؤریَن تام قٲیِم — اَکھ اِستِحکام یُس بیٚیِس کُنہِ تہِ تمدُنَن حٲصِل کٔرِتھ نہٕ۔ کِہَن زِ اَمہِ کٔر نہٕ طاقتہٕ سۭتۍ بلکہِ لوٗکَن ہٕنٛدِ عقیدٕ بدلاوِتھ اِختیار حٲصِل۔ فوجی طاقت چھِ عارضی — مگر عقیدٕ چھُ نسلن تام روزان۔

لوٗکھ ٲسۍ نہٕ یِتھۍ سوال پٔژھان زِ "کوٗنۍ چھُنہٕ خۄدایہ، کوٗنۍ چھُنہٕ دُنیا، تیٚلہِ کیٛازِ چھُس بہٕ کٲم کَران، کیٛازِ چھُس بہٕ یِم ڈھانٛچہٕ بناوان، کیٛازِ چھُس بہٕ زٔمیٖن کٲم کَران؟" اَمہِ بجایہِ ٲسۍ تِم یٔقیٖن تھاوان زِ "فِرعون چھُ موٗجوٗد، فرعون چھُ ونان، اَمہِ کِنۍ چھِ زٔمیٖن زِنٛدٕ، اَمہِ کِنۍ گژھِو اَسۍ زِنٛدٕ روزُن۔" زیٛادٕ سوال کٔرِتھ بغٲر، زیٛادٕ سوٗچِتھ بغٲر، تِمو کٔر تِتھۍ کٔنۍ یِتھۍ فرعونَن وون۔ اَمہِ کِنۍ اوس یہِ نِظام واریاہ مستحکم۔

غلامی چھنہٕ جسمٲنی زنجیرو سۭتۍ شروع گژھان - یہ چھ ذہنی زنجیرو سۭتۍ شروع گژھان۔ مصری ٲسۍ یہ سچٲیی پوٗرٕ پٲٹھۍ سمجان۔

آخنaten سُنٛد اِنقلاب — کنٹرول سِسٹمُک بہادُر مُقابلہٕ کَرن واجیٚنۍ

اَخیناتین سٕنٛز آمد - 1350 ق م

ہزارو وریو تام روٗد یہِ نِظام جٲری۔ مگر 1350 ق م کِس نزدیٖک آو اَکھ فرعون۔ تٔمِس اوس اَکھنٛاتن ناو۔ تٔمِس برٛونٛہہ ٲسۍ سٲری فرعون کٹھ پُتلی۔ پادٕرۍ ٲسۍ فرعونَس کنٹرول کَران۔ فرعون اوس صِرِف اَکھ گُڈٕلۍ۔ مگر 1350 ق م کِس نزدیٖک آو اَکھنٛاتن۔

اَخَنَتَن اوس یِتھ نِظامَس نِش آزاد گژھُن یَژھان۔ تٔمۍ کوٚر اَکھ سوال: "یِم پادٕرۍ کۄن چھِ؟ مےٚ کنٹرول کَرنہٕ خٲطرٕ؟ یہِ دۄن تہہَن ہُنٛد عِلمُک نِظام کیٛاہ چھُ؟ اَمہِ کھۄتہٕ زیٛادٕ، مےٚ گژھِ نہٕ یِمَن پادٕرٮ۪ن ہٕنٛدِ ذٔریعہٕ کنٹرول کَرنہٕ یِنۍ۔ بہٕ چھُس فرعون، بہٕ چھُس دیوتا، مےٚ گژھِ اِنتِظام کَرُن۔"

اِنقلاب — پادٕرۍ ہٹاوُن

امہ اناہس پیٹھٕ کوٚر اخناتینن اکھ انقلاب شروع۔ تٔمۍ کیٛاہ کوٚر:

تٔمہِ کٔرِ سٲرِوٕے پادٕرِ کڑِتھ ۔ تٔمہِ کٔرِ مندر تہٕ مزار بند ۔ تٔمہِ کٔرِ تِمَن مندرَن منٛز سٲرِوٕے دولت پنٛنِس اِختیارس منٛز ۔ اَمیُوک مطلب چُھ زِ تٔمہِ کٔرِ پادٕرِ ہٕنٛز اِقتِصٲدی طاقت تہِ کٔڈِتھ ۔ پتہٕ کٔرِ تٔمہِ اَکھ نوٚو اُصوٗل قٲیِم ۔

اکھ خۄدا — ایٹن

اَخَنَتَن سُنٛد اِعلان: صِرِف اَکھ دیوتا چھُ۔ اَمہِ دیوتا سُنٛد ناو چھُ "اَتَن"۔ اَتَن ہیٚکہِ آفتاب دیوتا سَمجِتھ۔ مگر ییٚتہِ چھُ اَکھ اَہَم نُکتہٕ۔ اَتھ دیوتا سٕنٛز چھِ نہٕ کانٛہہ اِنسانۍ شَکٕل۔ اَتَن چھُ اَکھ بےٚ شَکٕل، بےٚ مَشکل، بےٚ مَورَتھ کٲیناتی ہستی۔

یہِ چھُ اَکھ واریاہ اہم نُقطہٕ — کِیاہ کہِ اگر خۄدا سٕنٛز کانٛہہ اِنسٲنی شکل چھُنہٕ، تیٚلہِ چھُنہٕ سِیاسَتَس منٛز اِنسٲنی شِرکَت خٲطرٕ کانٛہہ جایہِ۔ کانٛہہ تہِ ہیٚکہِ نہٕ اَکھ اِنسان خۄدا سٕنٛدِس نُمٲیِندٕ سٕنٛدِ حٲیِثیتَس منٛز پیش کٔرِتھ تہٕ تٔمِس سٕنٛدِس ناوَس پٮ۪ٹھ سِیاسَت کٔرِتھ۔

اَمہِ علاوٕ، اَتھ دیوتاہَس تام واتنہٕ خٲطرٕ چھُنہٕ کُنہِ وُسیطَس ضروٗرت۔ کُنہِ پادٕرۍ سُنٛد ضروٗرت چھُنہٕ۔ کُنہِ کنٹرولر سُنٛد ضروٗرت چھُنہٕ۔ کانٛہہ تہِ شَخص ہیٚکہِ اَتھ دیوتاہَس سۭتۍ سیوٚدٕ رٲبطہٕ کٔرِتھ۔ یہِ اوس اَکھ اِعلان ییٚمۍ پوٗرٕ پادٕرۍ نِظامُک ریڑھٕ ہَڈ ژٔٹِتھ۔ کِیاہ کہِ پادٕرۍ سُنٛد پوٗرٕ اِختیار اوس "أسۍ چھِ دیوتاہَس سۭتۍ کَتھ کَران، أسۍ چھِ دیوتاہَس سۭتۍ سیوٚدٕ رٲبطہٕ تھاوان، اَتھ کِنۍ گژھِو تُہۍ أسۍ سٕنٛدِ ذٔریعہٕ یِنۍ" کَتھ پٮ۪ٹھ مبنی۔ اخنٛاتنَن کٔر سۄ کَتھ کٔڈِتھ۔

صِرِف مذہبی اِصلاحات نہٕ — سیاسی اِنقلاب

یہِ اوس 1350 قبل مسیح خٲطرٕ اکھ بُنیٲدی اِنقلاب۔ اَتھ گژھِ نہٕ صِرِف اکھ مذہبی اِصلاح وننہٕ یِنۍ۔ یہِ اوس اکھ مُکَمَل سِیٲسی اِنقلاب۔ اکھ اِنقلاب یُس کنٹرولرَن پوٗرٕ پٲٹھۍ تباہ کَرنہٕ خٲطرٕ زامُت اوس۔ اخیناطنَن گُزٲر پنٕنۍ زِنٛدگی یہِ عمل پھٲلاونَس منٛز۔

آخنaten سٕنٛز موت — پادٕرۍ چھِ واپس حملہٕ کران

مگر ییٚلہِ اَخیناتین موٗد، پادٕرِ آیہِ واپس۔ تِمو کوٚر اِختیار۔ تِمو کوٚر پوٗرٕ نِظام پنٛنِس اِختیارس منٛز واپس۔ تِمو کوٚر اَخیناتین سُنٛد پرٛٮ۪تھ نشان مِٹ۔ تِمو کوٚر تٔمِس سٕنٛدِ مُجسمہٕ پھٔٹراوِتھ۔ تِمو کوٚر تٔمِس سُنٛد ممی لَبنہٕ نامُمکن۔ تِمو کوٚر پوٗرٕ پٲٹھۍ سُہ شَہَر تباہ یُس اَخیناتینَن تعمیٖر کوٚرمُت اوس۔

اَکہِ نسلس منٛز، یُس اِنقلاب زاو — اَکھ پوٗرٕ خۄدایہ سُنٛد تصوٗر — آو پادٕرِیو دٔسۍ کامیٲبی سۭتۍ مارنہٕ۔ تِمو اوٚن گۄڈٕنِک نظام واپس۔ پادٕرِیو تہٕ کنٹرولرن ہٕنٛز پوٗرٕ کوٗشِش ٲس اَخیناتینس تٲریخس منٛزٕ پوٗرٕ پٲٹھۍ مِٹاونہٕ خٲطرٕ۔

مگر کینٛہہ نشانہٕ روُدِ ۔ امہِ كِنہِ چھٗ اسہِ از یہِ زانكٲری ۔

تُہۍ ہیٚکو عمارتہٕ مٲرِتھ — مگر خیالات نہٕ۔

پادٕرِ ہیٚکَن عمارتہٕ مِٹٲوِتھ۔ تِم ہیٚکَن شَہَر تباہ کٔرِتھ۔ تِم ہیٚکَن مُجَسَمَن پھٔٹٲوِتھ۔ مگر تِم ہیٚکَن نہٕ خیالات مارِتھ۔ "اَکھ دیوتا" ہُنٛد تصوٗر آو وارٕ وارٕ ژلِتھ۔ سُہ خیال ژلِتھ مصرٕ۔

توحیدٕچ پیدٲیِش — موسیٰ سۭتۍ رٲبطہٕ

یہِ "اکھ دیوتا" خیال گوٚو فلسطین، اسرائیل کِس علاقس تام۔ واریاہ چھِ مانان زِ موسیٰ ہن کٔر ییٚتہِ پٮ۪ٹھٕ شروٗع۔ ییٚتہِ پٮ۪ٹھٕ آو توحید، ابراہیمی عقیدٕ، تہٕ یہودیت پٲدٕ گژھِتھ۔

موزسَن کٔر بحرِ احمر الگ تہٕ مصر پٮ۪ٹھٕ ایشیاہَس منٛز دٲخٕل — اسرائیل کِس علاقس منٛز۔ اَتھ زٔمیٖنَس چھِ کنعان ونان۔ یہِ پوٗرٕ سفر، یہِ خیال — اوس اخناطن پٮ۪ٹھٕ زامُت، کیٚنٛہہ چھِ یِقیٖن کَران۔

دوبارٕ، یہِ چھُ قیاس آرٲیی۔ یِم چھِ نظریات۔ اَسۍ ہیٚکو نہٕ یقیٖنی طور زانِتھ۔ مگر یُس تہِ توحیدی مذہبی رٮ۪وایت موٗجوٗد چھُ، یُس تہِ رٮ۪وایتَن مغربی دُنیاہَس وۄتھراو، سُہ رٮ۪وایت چھُ مصرَس منٛز اَکھ اِنقلاب پٲدٕ گژھان۔ پادٕرۍ ہٕنٛدِس کنٹرولَس خِلاف، کنٹرول کَرن والٮ۪ن خِلاف۔

اَخیناتین سُنٛد سبق

اَخیناتین سٕنٛز دٔلیٖل چھےٚ نہٕ صِرِف اَکِس فرعون سٕنٛز دٔلیٖل۔ یہِ چھےٚ اَکِس بہادُر زَنہِ سٕنٛز دٔلیٖل ییٚمِس کنٹرول نِظامَس چیلنج کوٚر۔ تٔمۍ کوٚر پادٕرِ ہٕنٛدِس اِختیارس پٮ۪ٹھ سوال۔ تٔمۍ کوٚر ثالثی نِظام ختم۔ تٔمۍ کوٚر اَکِس دیوتا سُنٛد تصوٗر مُتعارِف۔ اَتھ دیوتاہَس اوس نہٕ کانٛہہ شکل، نہٕ کانٛہہ قَد — ییٚمہِ کِنۍ کانٛہہ تہِ ہیٚکہِ سیوٚدٕ دیوتاہَس سۭتۍ کَتھ کٔرِتھ۔ کانٛہہ پادٕرِ اوس نہٕ ضروٗری۔

مگر تٔمِس مرنہٕ پتہٕ آیہِ کنٹرولر واپس۔ تِمو کٔر اَخیناتینَس تٲریخَس منٛزٕ مِٹاونٕچ کوٗشِش۔ مگر تِم ہیٚکَن نہٕ خیالات مارِتھ۔ "اَکھ دیوتا" خیال گُزرِیو مصرٕ۔ سُہ خیال بنیوو اِبراہیمی مذہبَن ہُنٛد بُنیاد — یہوٗدِیَت، عیسٲیِیت، اِسلام۔ تِمَن مذہبَن کٔر مغربی دُنیاہَس تخلیق۔

مگر کنٹرولرن کٔر تِم مذہب تہِ پنُن۔ اخناطن سٕنٛدِس اِنقلابَس منٛز زامٕتۍ اِنقلابی طاقتہٕ آیہِ اَنٛدرٕ نِش دٲخٕل کَرنہٕ۔ یہِ عمل چھُ مصر پٮ۪ٹھٕ روم، روم پٮ۪ٹھٕ لندن، لندن پٮ۪ٹھٕ نیو یارک تام جٲری - اَز تہِ۔

یہِ چھُ "دُنیاہ کٮ۪ن کنٹرولرن ہُنٛد تٲریٖخ" سیریزُک ترٛیٚیِم مضمون۔ بیٚیِس مضمونَس منٛز پرٛیٚو بابلی قرضہٕ نِظام، یونٲنی فلسفہٕ، تہٕ کنٹرولرن ہٕنٛز دراندازی ہٕنٛز حکمت عملی۔
